// ======================= // 🧠 GAME STATE // ======================= let gameState = { paused: false, alive: true, hunger: 50, energy: 50, health: 100, happiness: 50, // ONLY driven by actions age: 0, ageProgress: 0, size: 1, poopReady: false, lastMealTime: 0, doctorCooldown: 0 }; // ======================= // 💾 SAVE / LOAD // ======================= function saveGame() { localStorage.setItem("tamagotchiSave", JSON.stringify(gameState)); } function loadGame() { const data = localStorage.getItem("tamagotchiSave"); if (data) gameState = JSON.parse(data); } loadGame(); // auto save setInterval(() => { if (!gameState.paused) saveGame(); }, 5000); // ======================= // ⏸️ PAUSE SYSTEM // ======================= function togglePause() { gameState.paused = !gameState.paused; } // ======================= // 🍎 FEED (affects hunger + happiness) // ======================= function feed() { if (!gameState.alive || gameState.paused) return; gameState.hunger = Math.max(0, gameState.hunger - 20); gameState.happiness = Math.min(100, gameState.happiness + 5); gameState.lastMealTime = Date.now(); } // ======================= // 💩 TOILET (only when needed) // ======================= function toilet() { if (!gameState.poopReady) return; gameState.poopReady = false; gameState.happiness = Math.min(100, gameState.happiness + 8); } // ======================= // 😴 SLEEP SYSTEM // ======================= let sleepInterval; function sleep() { if (!gameState.alive || gameState.paused) return; sleepInterval = setInterval(() => { gameState.energy = Math.min(100, gameState.energy + 2); gameState.happiness = Math.min(100, gameState.happiness + 1); }, 1000); } function wakeUp() { clearInterval(sleepInterval); } // ======================= // 🩺 DOCTOR SYSTEM // ======================= function doctor() { if (!gameState.alive || gameState.paused) return; if (gameState.doctorCooldown > 0) return; gameState.health = Math.min(100, gameState.health + 40); gameState.happiness = Math.max(0, gameState.happiness - 10); gameState.doctorCooldown = 30; } // ======================= // 🏀 MINI HOOP GAME // ======================= let hoopGame = { active: false, ballX: 0, basketX: 50 }; function startHoopGame() { hoopGame.active = true; let interval = setInterval(() => { if (!hoopGame.active || gameState.paused) { clearInterval(interval); return; } hoopGame.ballX = Math.random() * 100; if (Math.abs(hoopGame.ballX - hoopGame.basketX) < 10) { gameState.happiness = Math.min(100, gameState.happiness + 15); } else { gameState.happiness = Math.min(100, gameState.happiness + 2); } gameState.energy = Math.max(0, gameState.energy - 2); }, 800); } function endHoopGame() { hoopGame.active = false; } // ======================= // 🧬 AGE SYSTEM (1 YEAR = 10 MINUTES) // ======================= setInterval(() => { if (gameState.paused || !gameState.alive) return; gameState.ageProgress += 1; if (gameState.ageProgress >= 600) { gameState.age += 1; gameState.ageProgress = 0; gameState.size = 1 + gameState.age * 0.15; } }, 1000); // ======================= // ⏳ MAIN GAME LOOP // ======================= setInterval(() => { if (gameState.paused || !gameState.alive) return; // slow hunger + energy decay gameState.hunger = Math.min(100, gameState.hunger + 1); gameState.energy = Math.max(0, gameState.energy - 1); // poop after 1 minute if ( gameState.lastMealTime && Date.now() - gameState.lastMealTime > 60000 ) { gameState.poopReady = true; } // health system (neglect consequences) if (gameState.hunger > 80) gameState.health -= 1; if (gameState.energy < 10) gameState.health -= 1; if (gameState.health <= 0) gameState.alive = false; // doctor cooldown if (gameState.doctorCooldown > 0) { gameState.doctorCooldown -= 1; } updateUI(); }, 5000); // ======================= // 💔 MOOD IMPACT (no passive happiness decay) // ======================= setInterval(() => { if (gameState.paused || !gameState.alive) return; if (gameState.health < 30) { gameState.happiness = Math.max(0, gameState.happiness - 2); } }, 5000); // ======================= // 🖥️ UI UPDATE // ======================= function updateUI() { document.getElementById("hunger").innerText = gameState.hunger; document.getElementById("energy").innerText = gameState.energy; document.getElementById("health").innerText = gameState.health; document.getElementById("happiness").innerText = gameState.happiness; document.getElementById("age").innerText = gameState.age; // size scaling let pet = document.getElementById("pet"); if (pet) { pet.style.transform = `scale(${gameState.size})`; } // poop icon let poop = document.getElementById("poop"); if (poop) { poop.style.display = gameState.poopReady ? "block" : "none"; } }
